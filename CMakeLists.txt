cmake_minimum_required(VERSION 3.16)
project(BackNest VERSION 0.2 LANGUAGES CXX)
add_executable(backnest core/main.cpp)
target_compile_features(backnest PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(backnest PRIVATE /W4 /utf-8)
else()
  target_compile_options(backnest PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
  target_link_options(backnest PRIVATE -municode)
endif()
