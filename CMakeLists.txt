cmake_minimum_required(VERSION 3.16)
project(BackNest VERSION 0.2 LANGUAGES CXX)
option(BACKNEST_DESKTOP "Build Qt WebChannel desktop application" OFF)
add_executable(backnest core/main.cpp)
target_compile_features(backnest PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(backnest PRIVATE /W4 /utf-8)
else()
  target_compile_options(backnest PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BACKNEST_DESKTOP)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network WebChannel WebEngineWidgets)
  qt_standard_project_setup()
  qt_add_executable(backnest-desktop desktop/main.cpp)
  target_compile_features(backnest-desktop PRIVATE cxx_std_17)
  target_link_libraries(backnest-desktop PRIVATE Qt6::Widgets Qt6::Network Qt6::WebChannel Qt6::WebEngineWidgets)
  if(MSVC)
    target_compile_options(backnest-desktop PRIVATE /utf-8 /W4)
  endif()
endif()
if(MINGW)
  target_link_options(backnest PRIVATE -municode)
endif()
